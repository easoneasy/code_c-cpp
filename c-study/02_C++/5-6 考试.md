```
定义一个`Computer`类，包含私有数据成员：品牌（`char*`动态分配）、价格（`double`）。
```

```c++
class Computer {
public:
    // 构造函数
    Computer(const char* brand, double price)
    : m_brand(new char[strlen(brand) + 1]())
    , m_price(price) {
        strcpy(m_brand, brand);
    }
    // 析构函数
    ~Computer() {
        delete[] m_brand;
        m_brand = nullptr;
    }
    // 拷贝构造函数
    Computer(const Computer& rhs)
    : m_brand(new char[strlen(rhs.m_brand) + 1]())
    , m_price(rhs.m_price) {
        strcpy(m_brand, rhs.m_brand);
    }
    // 赋值运算符函数
    Computer& operator=(const Computer& rhs) {
        if (this != &rhs) {  // 处理自赋值
            delete[] m_brand;
            m_brand = new char[strlen(rhs.m_brand) + 1]();
            strcpy(m_brand, rhs.m_brand);
            m_price = rhs.m_price;
        }
        return *this;
    }
    
    // 打印信息
    void print() const {
        cout << "品牌：" << m_brand << "，价格：" << m_price << endl;
    }
    
private:
    char* m_brand;
    double m_price;
};

int main() {
    // 测试构造函数
    Computer pc1("Lenovo", 4999);
    cout << "pc1: ";
    pc1.print();
    
    // 测试拷贝构造函数
    Computer pc2 = pc1;  // 等价于 Computer pc2(pc1);
    cout << "pc2 (拷贝构造): ";
    pc2.print();
    
    // 测试赋值运算符
    Computer pc3("Dell", 5999);
    cout << "pc3 (赋值前): ";
    pc3.print();
    
    pc3 = pc1;  // 赋值
    cout << "pc3 (赋值后): ";
    pc3.print();
    
    // 测试自赋值
    pc3 = pc3;
    cout << "pc3 (自赋值): ";
    pc3.print();
    
    return 0;
}
```

```
实现一个单例模式的`Logger`类，用于记录日志。

要求：

1. 类中有一个`log`函数，接受一个字符串参数，输出该字符串（可简单打印到控制台）；
2. 确保整个程序中只有一个`Logger`实例（堆区实现）；
3. 提供`getInstance`静态方法获取实例，提供`destroyInstance`静态方法销毁实例；
4. 在`main`函数中测试单例的唯一性。
```

```c++
class Logger {
public:
    // 获取唯一实例
    static Logger* getInstance() {
        if (m_pInstance == nullptr) {
            m_pInstance = new Logger();
        }
        return m_pInstance;
    }
    // 销毁实例
    static void destroyInstance() {
        if (m_pInstance != nullptr) {
            delete m_pInstance;
            m_pInstance = nullptr;
        }
    }
    // 日志输出
    void log(const string& msg) {
        cout << "[LOG] " << msg << endl;
    }
private:
    // 私有化构造函数和析构函数
    Logger() {
        cout << "Logger实例创建" << endl;
    }
    ~Logger() {
        cout << "Logger实例销毁" << endl;
    }
    // 禁止拷贝和赋值
    Logger(const Logger&) = delete;
    Logger& operator=(const Logger&) = delete;
    // 静态实例指针
    static Logger* m_pInstance;
};

// 静态成员初始化
Logger* Logger::m_pInstance = nullptr;

int main() {
    // 测试单例唯一性
    Logger* logger1 = Logger::getInstance();
    Logger* logger2 = Logger::getInstance();
    
    cout << "logger1和logger2是" 
         << (logger1 == logger2 ? "同一个实例" : "不同实例") 
         << endl;
    
    // 使用单例
    logger1->log("程序启动");
    logger2->log("执行操作");
    
    // 销毁实例
    Logger::destroyInstance();
    
    // 再次获取实例（会重新创建）
    Logger* logger3 = Logger::getInstance();
    logger3->log("重新启动");
    
    Logger::destroyInstance();
    
    return 0;
}
```

使用`std::vector`存储多个学生信息（学生类为前面定义的`Student`类），实现以下功能：  
要求：  
1. 创建3个学生对象，存入`vector`；  
2. 使用增强for循环遍历并输出所有学生信息；  
3. 使用迭代器遍历并输出所有学生信息；  
4. 向`vector`末尾添加一个新学生，并再次输出。

```c++
class Student
{
public:
    //构造函数
    Student(const string name , int num ,double score)
    :m_num(num)
    ,m_score(score)
    {
        cout << "Student(string,int,double)" << endl;
        //字符串拷贝
        m_name = name;
    }
    //拷贝构造函数
    Student(const Student &rhs)
    :m_num(rhs.m_num)
     ,m_score(rhs.m_score)
    {
        m_name = rhs.m_name;
        cout << "Student(const Student &)" << endl;
    }
    //析构函数
    ~Student()
    {
        cout << "~Student" << endl;
    }
    //打印
    void print()
    {
        cout << m_name <<" "<< m_num << " " << m_score << endl;
    }

private:
    string m_name;
    int m_num;
    double m_score;
};

int main(int argc,char *argv[])
{
    Student s1{"zs",1,80};
    Student s2{"ls",2,90};
    Student s3{"ww",3,89};
    vector<Student> box;
    cout << box.size() << endl;
    cout << box.capacity() << endl;
    box.push_back(s1);
    cout << box.size() << endl;
    cout << box.capacity() << endl;
    box.push_back(s2);
    box.push_back(s3);
    cout << box.size() << endl;
    cout << box.capacity() << endl;
    //增强for循环遍历vector
    for(auto &element :box)
    {
        element.print();
    }
    cout << "-----------" << endl;
    //迭代器遍历vector
    auto itBegin = box.begin();
    for(;itBegin!=box.end();itBegin++)
    {
        itBegin->print();
    }
    cout << "-----------" << endl;
    //新增元素
    box.push_back({"zz",4,99});
    for(auto &element : box)
    {
        element.print();
    }
    cout << "-----------" << endl;
    auto itBgein1 = box.begin();
    for(;itBgein1!=box.end();++itBgein1)
    {
        itBgein1->print();
    }
    return 0;
}
```
