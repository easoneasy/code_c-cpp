内存分两类：内核区域和非内核（用户空间）

读文件，需要内核的参与。内核发挥了怎样的作用？

栈空间不属于内核
##### 有缓冲文件流

在用户空间中有一块缓冲区(4K左右)，数据会首先存在缓冲区内，读数据的时候才会进入自己的数组

加缓冲区的作用：从内核读取数据需要进行系统调用，效率很低。
加缓冲区—>防呆机制 
即使不知道缓冲区的设计目的，逐个fgetc 效率也不至于太差

返回值：FILE* 其中有大量指针，是用户空间的指针，**指向用户空间的地址**，不是内核的
![[有缓冲文件流.png]]
##### 无缓冲文件流

在用户空间没有缓冲区，数据直接从内核中读取，只提供了一个按照数组读取
![[无缓冲文件流.png]]
返回值：int类型（文件描述符） 
系统内核知道文件描述符的对应关系

使用open打开无缓冲区文件流 ，**返回int型**

open打开无缓冲区文件流：
有三个参数，第三个参数（权限）根据情况写
![[open属性.png]]

read不会填充任何额外的数据，读取文件里面有什么数据就会装入什么数据，不会和其他函数一样加\0，返回值是读取到的数据的个数
```c
#include <my_header.h>
int main(int argc,char *argv[])
{
    ARGS_CHECK(argc,2);
    int fd = open(argv[1],O_RDONLY);
    ERROR_CHECK(fd,-1,"fd open failed.\n");

    char buff[10]={0};
    //read 的返回值类型是ssize_t 需要long
    ssize_t ret = read(fd,buff,sizeof(buff));
    printf("ret = %ld\n",ret);
    printf("----%s----\n",buff);
    //1.txt的内容是123456789
    //ret =10 ,因为读取到文件中自带的换行符,占据了一个位置
    //----123456789    buff中有read读取到的换行符
    //----
    ssize_t ret2 = read(fd,buff,sizeof(buff));
    printf("ret2 = %ld\n",ret2);
    printf("----%s----\n",buff);
	//1.txt的内容是1234567890abcd
	//ret =5  ，读取到 abcd+换行符
	//----abcd   buff中有read读取到的换行符
    //67890----
    return 0;
}

```

将数据写入文件会自动加入一个换行符

read读取文件数据时，会读到换行符

write 表示实际写多少数据出去

read n个表示最多能装n个数据（数组的长度）, 实际能装多少取决于文件还剩多少

##### 复制文件

```c
#include <my_header.h>
int main(int argc,char *argv[])
{
    ARGS_CHECK(argc,3);

    int src_fd = open(argv[1],O_RDONLY);
    ERROR_CHECK(src_fd,-1,"src open failed.\n");
    int dest_fd = open(argv[2],O_WRONLY|O_CREAT|O_TRUNC,0764);
    ERROR_CHECK(dest_fd,-1,"dest open failed.\n");
    //初始化数组，存放read从文件中读取的数据
    char buff[100]={0};
    //存放read返回值，返回的是一次读取的数据个数
    ssize_t count =0;
    while((count = read(src_fd,buff,sizeof(buff)))!=0){
	    //从数组中向文件中写入数据，一次写入count个
        write(dest_fd,buff,count);

    }
    close(src_fd);
    close(dest_fd);
    return 0;
}

```

使用计时器，复制文件计时，比较使用两种不同大小的数组哪个快一点
```c
#include <my_header.h>
void copy_by_path(const char *src_path,const char *dest_path,
                  char *buff,int buff_count){

    int src_fd = open(src_path,O_RDONLY);
    int dest_fd = open(dest_path,O_WRONLY|O_CREAT|O_TRUNC);
    ssize_t count =0;
    while((count=read(src_fd,buff,buff_count))!=0){
        write(dest_fd,buff,count);
    }
    close(src_fd);
    close(dest_fd);

}

long long current_time_millis(void) {
    struct timespec ts;
    clock_gettime(CLOCK_REALTIME, &ts); // 获取当前时间
    return (long long)ts.tv_sec * 1000LL + ts.tv_nsec / 1000000; // 转换为毫秒
}
    
int main(int argc,char *argv[])
{
    ARGS_CHECK(argc,3);

    char *buff =(char *)malloc(20000);

    long long time1 = current_time_millis();
    copy_by_path(argv[1],argv[2],buff,1000);
    long long time2 = current_time_millis();
    copy_by_path(argv[1],argv[2],buff,4096);
    long long time3 = current_time_millis();
    
    printf("time2 - time1 = %lld\n",(time2-time1));
    printf("tiem3 - time2 = %lld\n",(time3-time2));

    return 0;
}

```
