
##### exec  
POSIX-C 函数
参数结束的标准  void *  0 (通用指针)  或者   NULL
适用于fork之后，在当前进程的上下文中加载并运行一个新的程序
execl 参数是散的
execv参数可以是给定一个指针数组，每个元素指向同一个字符串的不同位置
执行成功，不会返回原来的程序，原执行程序已经被新程序替换
有返回值说明出现错误，返回-1

在使用exec的时候，参数是什么就是什么
为了保持shell调用和execl调用的一致，将execl的第一个参数写成path
```c
pid_t ret = fork();
if(ret ==0){
	printf("child process."\n);
	execl("./test_caculate","./test_caculate","6","7",(void *)0);
}else{
	printf("father process.\n");
	sleep(30);
}
```

```c
---test_caculate.c---
int num1;
int num2;
sscanf(argv[1],"%d",num1);
sscanf(argv[2],"%d",num2);
printf("num1 + num2 = %d",num1+num2);
```
shell进程本质就是fork+exec
一直等待输入

##### 孤儿进程：
父进程先结束，子进程不结束，这时候子进程就成为了孤儿进程
孤儿进程会被pid=1的进程收养
使用 ctrl + c 无法结束整个孤儿进程，可以使用kill pid命令关闭这个进程
```c
pid_t fork_ret = fork();
    if(fork_ret==0){
        printf("child process.\n");
        while(1){
            printf("child process is running.\n");
            sleep(1);
        }
    }else{
        printf("father process.\n");
        sleep(5);
        printf("father sleep over.\n");
    }

```

##### 僵尸进程：
子进程先结束，父进程不结束。
因为父进程无法为其处理数据，导致子进程无法退出。Z状态
按ctrl + c 结束父进程，子进程也会跟着退掉
```c
pid_t ret = fork();
    if(ret == 0){
        printf("child process.\n");
        sleep(3);
        printf("child process over.\n");
    }else{
        printf("father process.\n");
        while(1){
            printf("father process is running.\n");
            sleep(2);
        }
    }
```

##### wait
父进程等子进程（阻塞函数）
不知道等谁，谁先执行好等谁
一定要有子进程才用wait，否则就是-1
等待成功返回值是：子进程的pid
等待失败返回值就是-1
//父进程等子进程分别执行下载任务，最终由父进程汇总，需要等待子进程
```c
pid_t ret = fork();
    if(ret == 0){
        printf("child process.\n");
        sleep(10);
        printf("child process over.\n");
    }else{
        printf("father process.\n");
        sleep(2);
        printf("father sleep over.\n");
        pid_t wait_pid = wait(NULL);
        //获取wait的返回值，得到子进程的pid
        printf("father process over.wait pid = %d\n",wait_pid);
    }
```


wait的入参，
wstatus  整型指针   int * （传入传出参数） 
相当于得到子进程的状态（子进程）
正常退的退出码，是return的结果
不正常退的信号标志，是kill 的选项 kill -2  就会拿到2 kill -9 会拿到9
退的是子进程
```c
if(fork()==0){
        printf("child process.\n");
        sleep(20);
        printf("child process sleep over.\n");
        return 0;
    }else{
        printf("father process.\n");
        int wstatus;
        pid_t wait_ret = wait(&wstatus);
        printf("father process over.wait ret = %d\n",wait_ret);

        if(WIFEXITED(wstatus)){
            printf("child status : %d\n",WEXITSTATUS(wstatus));
        }else if(WIFSIGNALED(wstatus)){
            printf("child signed:%d\n",WTERMSIG(wstatus));
        }
    }
```

###### waitpid
可以指定等待的进程，可以把fork的返回值（子进程的id）给waitpid参数
最后一个参数可以让wait不阻塞，默认是0，阻塞
WNOHANG 可以让wait不阻塞
```c
pid_t fork_ret = fork();
if(fork_ret == 0){
    printf("child process begin.\n");
    sleep(20);
    printf("child process over.\n");
}else{
    printf("father process begin.\n");
    pid_t waitpid_ret = waitpid(fork_ret,NULL,WNOHANG);
    printf("wait over,ret = %d\n",waitpid_ret);
    printf("child process over.\n");
}
```




进程
有自己的id、也有组id
组id默认是跟外面的shell一起的，启动的时候会设置为自己的pid
组可以用于一次控制多个进程
子进程由父进程fork出来的，所以子进程和父进程一个组
```c
getpgrp();
//与程序被谁拉起来的相关
```


##### setpgid
让进程脱离当前组
```
setpgid {
pid    0表示当前组
pgid  0表示当前进程PID作为进程组ID
}
setpgid(0,0)
//让当前进程成为一个新的进程组的组长
```

```c
printf("pid = %d,pgrp = %d\n",getpid(),getpgrp());
    if(fork()==0){
        setpgid(0,0);
        //子进程自己一个组，成为新进程组的组长
        printf("child process pid = %d,pgrp = %d\n",getpid(),getpgrp());
    }else{
        printf("father process pid = %d,pgrp = %d\n",getpid(),getpgrp());
    }
```

##### setsid
创建一个新的会话（脱离当前会话）
使调用进程成为此会话的首进程
使用ctrl + c 停止不了该进程，当前会话关掉不会影响到这个进程
只能用kill杀死
```c
if(fork()==0){
    printf("child process begin.\n");
    setsid();
    //子进程成为新会话的首进程
    while(1){
        printf("child process running.\n");
        sleep(1);
    }
}else{
    printf("father process begin.\n");
    sleep(60);
    printf("father process over.\n");
}
```
