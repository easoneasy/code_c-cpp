
进程是内存拥有的基本单位（资源分配），进程切换的时候会进行上下文切换，效率比较低

一个进程中的线程会共享进程的资源，（数据段、堆、代码段、内核）。线程有自己的用户态栈空间。
线程是调度执行的基本单位，每个线程都有自己的task_struct结构体

| 对比项 | 进程（Process）            | 线程（Thread）                 |
| --- | ---------------------- | -------------------------- |
| 本质  | 资源分配的基本单位              | CPU 调度执行的基本单位              |
| 资源  | 有独立的地址空间、文件描述符、信号处理等   | 共享所属进程的所有资源，只独有栈、寄存器、线程 ID |
| 开销  | 创建、销毁、切换开销大            | 创建、切换、销毁开销很小               |
| 通信  | 进程间通信复杂（管道、消息队列、共享内存等） | 线程间直接读写全局变量即可，通信简单         |
| 影响  | 一个进程崩溃，其他进程不受影响        | 一个线程崩溃，整个进程都会崩溃            |
| 关系  | 一个进程可以包含多个线程           | 线程必须依附进程存在，不能独立运行          |
| 安全性 | 进程间相互独立，更安全            | 多线程竞争全局变量，容易出现并发问题         |

##### pthread_create-堆上的变量
主线程申请堆空间，子线程进行修改，也可以进行数据共享
线程传参（第4个参数）可以将值转递给线程入口函数

```c
//接收void *参数
void *thread_func(void *arg){
    printf("child thread.\n");
    //将void *强转成char*
    //通过修改指针指向地址的内容来更改，主线程可以看到
    char *p = (char *)arg;
    strcpy(p,"hello");
    return NULL;
}
int main(int argc,char *argv[]){
//malloc申请堆空间，返回的是void *，需要强转类型
    char *ptmp=(char *)malloc(10*sizeof(char));

    pthread_t pthread_id;
    //第四个参数需要使用void *传递
    int ret = pthread_create(&pthread_id,NULL,thread_func,(void *)ptmp);
    THREAD_ERROR_CHECK(ret,"thread create failed.\n");
    printf("main thread.\n");

    sleep(3);
    printf("ptmp : %s\n",ptmp);
    free(ptmp);
    ptmp=NULL;
    return 0;
}
```

##### pthread_join
捕获线程的退出   是阻塞函数吗？是
等待其他线程的结束

第二个参数   目标线程的退出状态，指向线程的退出时携带参数
```c
void *thread_func(void *arg){
    int *p=(int *)malloc(4);
    *p=30;
    return (void *)p;
}
int main(int argc,char *argv[]){
    pthread_t pthread_id;
    int ret = pthread_create(&pthread_id,NULL,thread_func,NULL);
    THREAD_ERROR_CHECK(ret,"thread create faild.\n");
    //一级指针存放目标程序的返回值（退出状态）
    void *tmp;
    ret = pthread_join(pthread_id,&tmp);
    THREAD_ERROR_CHECK(ret,"thread join.\n");
    printf("thread join over.\n");
    int *p=(int *)tmp;
    printf("thread join p = %d\n",*p);
    return 0;
}
```

二级指针定义的时候需要申请空间（malloc）
还需要释放吗？ 需要，释放的顺序是先释放子线程的堆空间，再释放主线程的堆空间
```c
void *thread_func(void *arg){
    printf("child thread.\n");
    int *p=(int *)malloc(10);
    *p=50;
    return (void *)p;
}
int main(int argc,char *argv[]){
    pthread_t pthread_id;
    int ret = pthread_create(&pthread_id,NULL,thread_func,NULL);
    THREAD_ERROR_CHECK(ret,"thread create failed.\n");
    
    void **ptmp=(void **)malloc(10);
    ret = pthread_join(pthread_id,ptmp);
    THREAD_ERROR_CHECK(ret,"thread join failed.\n");
    printf("thread join over.\n");

    int retval = *(int *)*ptmp;
    printf("retval=%d\n",retval);
    //先释放子线程的堆空间再释放主线程
    free(*ptmp);
    *ptmp=NULL;
    free(ptmp);
    ptmp=NULL;
    return 0;
}
```
局部变量通过取地址传递，不可以
函数调用结束，局部变量销毁
可以把局部变量写成 static + 局部变量（生命周期和整个进程一致）
```c
void *thread_func(void *arg){
    printf("child thread.\n");
    /* int *num = (int *)malloc(10); */
    /* *num = 50; */
    //加了static之后生命周期和进程一样长，就不会被销毁
    static  int num =100;
    return (void *)&num;
    //局部变量的地址，函数结束之后局部变量销毁
}
int main(int argc,char *argv[]){
    pthread_t thread_id;
    int ret = pthread_create(&thread_id,NULL,thread_func,NULL);
    THREAD_ERROR_CHECK(ret,"thread create failed.\n");

    void *retval;
    ret = pthread_join(thread_id,&retval);
    THREAD_ERROR_CHECK(ret,"thread join failed.\n");
    printf("thread join over.\n");

    int *ptmp = (int *)retval;
    printf("ptmp = %d\n",*ptmp);
    return 0;
}
```

可以获取堆区、long型、全局数据这些数据都没有问题，如果获取的是子线程的局部变量的地址时就会出现问题

##### 线程的退出
return  退出函数
等待线程执行完成之后就会退出
```c
void func(){
    printf("func\n");
    //返回调用它的函数thread_func
    return ;
}
void *thread_func(void *arg){
    printf("child thread.\n");
    func();
    printf("child thread func over.\n");
    //返回调用它的函数
    return NULL;
}
int main(int argc,char *argv[]){
    pthread_t thread_id;
    int ret = pthread_create(&thread_id,NULL,thread_func,NULL);
    THREAD_ERROR_CHECK(ret,"thread create failed.\n");
    printf("main thread.\n");
    ret = pthread_join(thread_id,NULL);
    THREAD_ERROR_CHECK(ret,"thread join failed.\n");
    printf("main thread over.\n");
    return 0;
}
```

###### pthread_exit();  
线程退出的方式，只要运行这个，后面的语句无法执行。与return不同。
参数可以将线程的退出状态传递，另一个线程使用pthread_join(pthread_id,&retval);接收
```c
void func(){
    printf("func\n");
    //子线程执行结束，直接返回主线程
    pthread_exit(NULL);
    printf("func over.\n");
}
void *thread_func(void *arg){
    printf("child thread.\n");
    func();
    printf("child thread func over.\n");
    return NULL;
}
int main(int argc,char *argv[]){
    pthread_t thread_id;
    int ret = pthread_create(&thread_id,NULL,thread_func,NULL);
    THREAD_ERROR_CHECK(ret,"thread create failed.\n");

    printf("main thread.\n");
    ret = pthread_join(thread_id,NULL);
    THREAD_ERROR_CHECK(ret,"thread join failed.\n");
    printf("main thread over.\n");
    return 0;
}
```

退出函数的返回值可以用pthread_join的第二个参数来接收
```c
void func(){
    printf("func \n");
    long num =100;
    pthread_exit((void *)num);
    //返回一个(void *)的返回值

}
void *thread_func(void *arg){
    printf("child thread.\n");
    func();
    printf("child thread over.\n");
    return NULL;
}
int main(int argc,char *argv[]){

    int ret = pthread_create(&pthread_id,NULL,thread_func,NULL);
    THREAD_ERROR_CHECK(ret,"thread create failed.\n");
    printf("main thread.\n");
    
    void *num;
    ret = pthread_join(pthread_id,&num);

    long p = (long)num;
    printf("pthread exit retval=%ld\n",p);
    return 0;
}
```
