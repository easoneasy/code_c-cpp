
中断的对象是CPU，**信号的对象是进程**
中断涉及到的是, CPU对紧急事件的处理和响应, 以及进程的调度和上下文切换;
而信号则是被某些行为产生, 产生的信号会修改对应进程的状态, 对应的进程, 又根据被改变的进程状态, 做出对应反应。


信号的种类  ：使用kill -l命令查看
1-31 ：标准信号   不会排队
34-64：实时信号  会排队，放入队列中

常用的5个信号：
2 、 3、9、18、19
3号是生成转储文件，18暂停进程，19取消暂停进程

##### sigaction结构体
1、第一个成员被默认执行
2、如果想使用sa_sigaction需要将flags配合SA_siginfo使用
3、sa_flags这个标志位可以设置为多种形式
4、sa_mask掩码，可以在触发某个信号过程中，忽略（阻塞）其他信号，处理完此信号之后再处理其他信号

memset将act里的内容清空设为0

```c
void fun(int num){
    printf("num : %d\n",num);
}
int main(int argc,char *argv[]){
	
    struct sigaction act,oldact;
    //清空act
    memset(&act,0,sizeof(act));
    //信号的新处理动作是执行fun函数
    act.sa_handler=fun;
    //要操作的信号编号，指定信号的新处理动作，获取信号的上一个处理动作
    sigaction(2,&act,&oldact);
    printf("begin while.\n");
    while(1);
    return 0;
}
```
//相当于
```c
signal(2,fun);
```

C语言中，数组名是数组的入口地址
函数名是函数的入口地址

##### sa_flags
选项
- SA_SIGINFO：使用 sa_sigaction成员而不是 sa_handler作为信号处理函数。
- SA_RESETHAND: 处理完捕获的信号以后，信号处理回归到默认( 一次注册只生效一次)
- SA_NODEFER: 在信号处理函数执行期间，同一个信号设置可以再次被触发
- SA_RESTART：使被信号打断的系统调用自动重新调用。

act.sa_flags=SA_SIGINFO;与act.sa_sigaction=func;一起使用
```c
void func(int num ,siginfo_t *siginfo,void *p){
    printf("num : %d\n",num);
    printf("func pid :%d\n",getpid());
    printf("xxxpid:%d\n",siginfo->si_pid);
}
int main(int argc,char *argv[]){
    struct sigaction act,oldact;
    memset(&act,0,sizeof(act));
    act.sa_sigaction=func;

    act.sa_flags=SA_SIGINFO;
    sigaction(2,&act,&oldact);

    printf("begin while\n");
    while(1);
    return 0;
}
```

SA_RESETHAND：一次注册只生效一次

sa_mask 的 sigset_t类型是一个位图，存放阻塞集合
sa_mask 在执行2号信号的过程中执行3号，不会被立马中断，会在2号执行完成之后再执行3号。
（临时屏蔽）
```c
void fun1(int num){
    printf("fun1 sleep begin.\n");
    sleep(10);
    printf("fun1 sleep over.\n");
}
int main(int argc,char *argv[]){
    struct sigaction act,oldact;
    memset(&act,0,sizeof(act));
    act.sa_handler=fun1;
    
    sigset_t mask;
    sigemptyset(&mask);
    sigaddset(&mask,3);
    act.sa_mask=mask;
    
    sigaction(2,&act,&oldact);
    printf("while begin.\n");
    while(1);
    return 0;
}
```

##### sigpending
检测某些信号是否在未决信号集中
读取内核的未决信号集，复制到 set
```c
void fun1(int num){
    printf("num : %d\n",num);
    printf("sleep begin.\n");
    sleep(10);
    sigset_t set;
    sigemptyset(&set);
    //读取内核的未决信号集，复制到 set
    sigpending(&set);
    //判断set里有没有3号信号
    if(sigismember(&set,3)){
        printf("3 is inside.\n");
    }else{
        printf("3 not inside.\n");
    }
    printf("sleep over.\n");
}
int main(int argc,char *argv[]){
    struct sigaction act,oldact;
    memset(&act,0,sizeof(act));
    act.sa_handler=fun1;
    
    sigset_t mask;
    sigemptyset(&mask);
    sigaddset(&mask,3);
    act.sa_mask=mask;

    sigaction(2,&act,&oldact);
    printf("while begin.\n");
    while(1);
    return 0;
}

```

##### sigprocmask 
检查和更改进程的信号屏蔽字（信号掩码mask）
信号掩码确定了哪些信号可以递送给该进程，哪些信号被阻塞
sigprocmask本身不是阻塞函数
与sigaction的sa_mask不同的是 sa_mask是临时屏蔽，而这个是**全程屏蔽**
SIG_BLOCK：把set内信号添加阻塞
SIG_UNBLOCK：解除set内信号阻塞
```c
void func(int num){
    printf("num : %d\n",num);
}
int main(int argc,char *argv[]){
    signal(2,func);
    
    sigset_t set,oldset;
    sigemptyset(&set);
    sigaddset(&set,2);
	//set内的2号信号添加阻塞
    sigprocmask(SIG_BLOCK,&set,&oldset);
    sleep(10);
    printf("sleep over.\n");
    //set内的2号信号解除阻塞
    sigprocmask(SIG_UNBLOCK,&set,&oldset);

    printf("while begin.\n");
    while(1);
    return 0;
}
```

##### kill 
给另一个进程发送信号
```c
kill(pid,sig)
//进程id、信号数值
```
父进程把信号发送给子进程之后就会结束，
子进程此时还会运行（因为2信号被转成了fun函数，不会起到停止进程的作用）
所以子进程变成了孤儿进程
```c
void func(int num){
    printf("num : %d\n",num);
}
int main(int argc,char *argv[]){
    pid_t pid = fork();
    if(pid==0){
        printf("child process.\n");
        //子进程接收到2信号会调取fun函数
        signal(2,func);
        while(1){
            sleep(1);
            printf("while is running.\n");
        }
    }else{
        printf("father process sleep begin.\n");
        sleep(10);
        //给子进程发送2信号
        kill(pid,2);
    }
    return 0;
}
```

##### sigprocmask
在sleep时按ctrl + c不会响应，sleep之后会响应
是因为sleep之后sigprocmask解除了屏蔽，才会响应
之后又到pause阻塞，等待信号输入
```c
void func(int num){
    printf("num = %d\n",num);
}
int main(int argc,char *argv[]){
    signal(2,func);

    sigset_t set,oldset;
    sigemptyset(&set);
    sigaddset(&set,2);

    sigprocmask(SIG_BLOCK,&set,&oldset);
    printf("sleep begin.\n");
    sleep(10);
    printf("sleep over.\n");

    sigprocmask(SIG_UNBLOCK,&set,&oldset);

    printf("pause begin.\n");
    pause();
    printf("pause over.\n");

    printf("while begin.\n");
    while(1);

    return 0;
}
```


##### sigsuspend
更改进程的信号屏蔽字并挂起进程的执行，直到捕获到一个信号
sigsuspend = 「临时改屏蔽集」 + 「pause 等信号」 + 「自动恢复屏蔽集」 
它的完整动作是：
保存当前进程的信号屏蔽字
把信号屏蔽字改成你传入的 mask
阻塞等待一个信号到来（像 pause 一样）
信号来了→处理完→自动恢复原来的屏蔽字

ps：在sleep时按ctrl +c 不会响应，但是sleep之后就会马上输出func，并且sigsuspend不会阻塞，
是因为sleep时屏蔽了2，在屏蔽期间，2变成未决状态，一旦屏蔽解除，内核会立马触发这个信号，执行处理函数
在睡完之后，将屏蔽字暂时设置成了3，此时2会触发sigsuspend，之后屏蔽信号又成为了2
```c
void func(int num){
    printf("num = %d\n",num);
}
int main(int argc,char *argv[]){
//将2信号设置为屏蔽信号
    signal(2,func);
    sigset_t set,oldset;
    sigemptyset(&set);
    sigaddset(&set,2);
    sigprocmask(SIG_BLOCK,&set,&oldset);

    sleep(5);
    printf("sleep over.\n");
//sleep之后将3信号作为屏蔽信号
    sigset_t mask;
    sigemptyset(&mask);
    sigaddset(&mask,3);
    printf("sigsuspend begin.\n");
//这里会阻塞，等待一个非3信号，就会解除3信号的屏蔽
    sigsuspend(&mask);
//恢复一开始设置的屏蔽字2信号
    printf("while begin.\n");
    while(1);
    printf("while over.\n");
    return 0;
}

```


##### alarm函数
设置定时器，指定的秒数到期时，内核会向该进程发送SIGALRM信号
```c
void func(int num){
    printf("func num = %d\n",num);
    //内核向此进程发送SIGALRM信号，会传给signal函数
    //一秒就会发送一次
    alarm(1);
}
int main(int argc,char *argv[]){
//收到SIGALRM信号就会执行func函数
    signal(SIGALRM,func);
    //计时器到期时，内核会向该进程发送SIGALRM信号
    //5秒之后发送
    alarm(5);
    printf("while begin.\n");
    while(1);
    printf("while over.\n");
    return 0;
}
```

##### setitimer
高精度定时器

which参数：指定定时器的类型。常用的类型包括：

-ITIMER_REAL：按照真实时间,  当时间到达,  发出一个 SIGALRM 信号。
-ITIMER_VIRTUAL：按照用户态代码执行时间计算, 当时间到达,  发出一个 SIGVTALRM信号。
-ITIMER_PROF：按照用户态用户态和内核态代码执行时间计算, 当时间到达,  发出一个 SIGPROF 信号。
```c
int setitimer(
// 定时器的类型
    int which, 
// 指定的新的定时器值
    const struct itimerval *new_value, 
// 存储定时器的前一个值
    struct itimerval *old_value 
);
//返回值：成功0，失败-1
```

```c
void func(int num){
    printf("num : %d \n",num);
}
int main(int argc,char *argv[]){
    signal(SIGALRM,func);
    struct itimerval new_value;
//循环间隔时间（闹钟响之后，每隔多久响一次）
    new_value.it_interval.tv_sec=1;
    new_value.it_interval.tv_usec=0;
//第一次延迟时间（程序启动后，多久第一次响）
    new_value.it_value.tv_sec=10;
    new_value.it_value.tv_usec=0;

    setitimer(ITIMER_REAL,&new_value,NULL);

    printf("while begin.\n");
    while(1);
    printf("while over.\n");
    return 0;
}
```
