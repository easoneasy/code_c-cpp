//write一条一条写数据，read一条一条读数据，
//write睡的时候，管道中的数据已被read读完，造成read读空阻塞
read:
	//read端打开管道的时，需要write端也要打开管道

	int read_p =open("1.pipe",O_RDONLY);
    ERROR_CHECK(read_p,-1,"read open failed.\n");
    char buff[4096]={0};
    for(int i=0;i<100;i++){
        printf("read begin.\n");
        ssize_t ret = read(read_p,buff,sizeof(buff));
        printf("round----%d----total_num:%ld\n",i,ret);
        printf("read end.\n");
    }
    close(read_p);
----------------------------------------------------------------------
write:
    int write_p =open("1.pipe",O_WRONLY);
    ERROR_CHECK(write_p,-1,"write open failed.\n");
    char buff[4096]={0};
    for(int i=0;i<10000;i++){
        printf("write begin.\n");
        ssize_t ret = write(write_p,buff,sizeof(buff));
        printf("round---------%d--------total_num:%ld\n",i,ret);
        printf("write end.\n");
        //睡10s
        sleep(10);
    }
    close(write_p);

关闭一端，另一端会怎样？
写端关，读端继续把数据读完，不会阻塞，每次读出来的数据为0
读端关，写端继续往管道中写数据，写端收到退出的信号，退出程序


双工：
先read再write，因为读空阻塞，所以会出现程序卡死
先write再read，数据会先存储到管道中，read就能读出数据 

read、write的length语义是不一样的
read：一个是最多能装多少
write：一个是实际要写出去多少

使用select监听通信
```c
printf("open 1.pipe begin.\n");
    int write_p = open("1.pipe",O_WRONLY);
    printf("open 1.pipe over.\n");
    ERROR_CHECK(write_p,-1,"open 1.pipe failed.\n");
    printf("open 2.pipe begin.\n");
    int read_p = open("2.pipe",O_RDONLY);
    printf("open 2.pipe over.\n");
    ERROR_CHECK(read_p,-1,"open 2.pipe failed.\n");

    while(1){
        fd_set set;
        FD_ZERO(&set);
        FD_SET(read_p,&set);
        FD_SET(STDIN_FILENO,&set);
 
        int ret = select(read_p+1,&set,NULL,NULL,NULL);
        if(FD_ISSET(STDIN_FILENO,&set)){
            printf("stdin enabled.\n");
            char buff1[100]={0};
            ssize_t ret = read(STDIN_FILENO,buff1,sizeof(buff1));
            printf("ret = %ld\n",ret);
            printf("stdin over.\n");
            printf("write begin.\n");
            write(write_p,buff1,ret);
            printf("write over.\n");
        }
        if(FD_ISSET(read_p,&set)){
            printf("read 2.pipe enabled.\n");
            char buff2[100]={0};
            ssize_t count = read(read_p,buff2,sizeof(buff2));
            printf("read 2.pipe over.\n");
            if(count ==0){
                printf("userb closed.\n");
                break;
            }
            printf("msg from userb: %s\n",buff2);
         }
    }
    close(write_p);
    close(read_p);
```


作业：
编写程序A和B。A负责将文件的名字、长度和内容通过管道发送B，B需要新建一个目录/文件，并将该文件存储起来。
比如发送 /home/zhou/test1.txt 
即使目录不存在，也能新建目录，新建文件。并且将内容弄进去

发送顺序要和接收顺序相同
read 第三个参数是想要读几个字节。write同理
发送、接收文件名、文件的长度是int类型，只需要读一个int的大小即可。所以read的第三个参数应该是sizeof(int)
发送文件名、文件内容是字符串，需要接收的长度是strlen的长度，而不是装内容数组的长度。
```c
#include <my_header.h>
int main(int argc,char *argv[]){
   
    int writep = open("1.pipe",O_WRONLY);
    ERROR_CHECK(writep,-1,"open pipe failed.\n");

    char file_name[]="test1.txt";
    int file_name_len=strlen(file_name);

    write(writep,&file_name_len,sizeof(int));
    write(writep,file_name,file_name_len);

    char file_content[]="hello world!";
    int file_len = sizeof(file_content);
    int str_len = strlen(file_content);
    
    write(writep,&file_len,sizeof(int));
    write(writep,&str_len,sizeof(int));
    write(writep,file_content,str_len);

    close(writep);

    return 0;
}
```

    mkdir("files",0777);
    chdir("files");

    int new_fd = open(file_name,O_RDWR|O_CREAT,0666);
    ftruncate(new_fd,file_len);
    write(new_fd,file_content,str_len);
    close(readp);
    close(new_fd);

    return 0;
}
